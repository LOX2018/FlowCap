### 1.3 录制接口签名

`recorder.py` 提供两条入口：

**① CLI**（最常用）：
```bash
python -m replay.recorder --from <path> --name <样本名> \
    --desc "说明" --provenance "来源" \
    --account-uid <uid> \
    --expect k=v [--expect k=v ...]
```

**② Python API**（供构造型夹具使用）：
```python
from replay import recorder

# 从文件录制
recorder.record(src="/path/to/sample.bin", name="sample_name",
    desc="...", provenance="...", account_uid="...",
    expected={"k": v},
    update=False)

# 从内存字节录制（构造型夹具）
recorder.record_blob(blob=b"...", name="sample_name",
    desc="...", provenance="...",
    account_uid=None, expected=None)
```

每次录制自动计算 sha256 + 写入 manifest —— 样本一旦冻结即不可变（sha256 门禁）。

## 3. 样本①：cmd301 历史响应（get_by_conversation）

### 3.1 来源

抖音 IM API `POST https://imapi.douyin.com/v1/message/get_by_conversation` 的 HTTP 响应原始字节。

当前已经存在 hook 捕获路径：BCC 注入的 `CAP_WP_MESSAGE_HOOK_JS` 正则匹配 `get_by_conversation` 并推入 `window.__CAP_WP_MESSAGE__.events`，后端 `recv_daemon` 轮询读取后由 `wp_recv` 解析。

### 3.2 格式

protobuf 二进制，解析结构（经 blackboxprotobuf 解码后）：
```python
response['6']['301'] = {
    '1': [msg1, msg2, ...],    # 本页消息数组
    '5': {                      # 分页信息
        '1': 本页起始序号,
        '2': 会话总条数,
        '3': 下一页游标 / None（无更多页）,
        '4': 本页结束 ts,
        '5': short_id,
        '6': conv_id,
    }
}
```

每条消息的 content（`m['8']`）是 JSON 字符串，内含：
```json
{
  "text": "...",
  "resource_url": {
    "skey": "<64 hex chars = 32B AES-256 密钥>",
    "origin_url_list": ["<加密原图 URL>"],
    "large_url_list": [...]
  }
}
```

`skey` + `origin_url` 由 `_extract_image_secret()` 提取，用于图片解密回归验证。

### 3.3 真机录制步骤

**前置条件**：BCC 在线，目标账号已登录抖音私信页。

```
Step 1: 在 BCC 所在浏览器打开 douyin.com/chat
Step 2: 确认 WP hook 已注入 (window.__CAP_WP_MESSAGE__ 已创建)
Step 3: 点击一个较长的对话（确保 >50 条消息，触发分页）
Step 4: 在该对话中持续滚动到最早消息（触发多次 get_by_conversation）
Step 5: 从 BCC 日志中提取 get_by_conversation HTTP 响应
        日志关键词: "imapi.douyin.com.*get_by_conversation" 或
        wp_recv 的 process_events 处理条目
```

**替代方案**（更可控）：通过 API 直接拉取。

```python
# 在真机环境下运行（有 valid auth）：
from auto_dm import accounts as acc
from dy_apis.douyin_api import DouyinAPI
from builder.header import HeaderBuilder, HeaderType
from builder.params import Params
import requests

auth = DYLoginApi._load_auth_from_env(acc.env_path_of("your_account"))
cid = "0:1:uid_a:uid_b"  # 从已录 init_packet 中选一个含图片消息的会话
short_id = None

# 构造 cmd301 请求
request = ProtoBuilder.build_normal_request(auth, 301)
body_bytes = request.SerializeToString()
inner = _build_301_body(cid, short_id, cursor=0, count=200)
tag = _pb_varint((301 << 3) | 2)
payload = tag + _pb_varint(len(inner)) + inner
body_bytes = body_bytes + b"\x42" + _pb_varint(len(payload)) + payload

resp = requests.post(
    "https://imapi.douyin.com/v1/message/get_by_conversation",
    headers=HeaderBuilder().build(HeaderType.PROTOBUF).get(),
    cookies=auth.cookie,
    data=body_bytes,
)

# 保存原始响应字节
with open("cmd301_response.bin", "wb") as f:
    f.write(resp.content)
```

**推荐**：选一个含图片消息的会话（skey/origin_url 非空），且消息数 >50 以覆盖分页。

```python
class TestCmd301Replay(unittest.TestCase):
    """cmd301 历史补全回归（含图片解密要素）。"""

    FIXTURE = "cmd301_history"

    @classmethod
    def setUpClass(cls):
        if cls.FIXTURE not in loader.list_fixtures():
            raise unittest.SkipTest(f"尚未录制 {cls.FIXTURE} 样本")
        cls.entry = loader.describe(cls.FIXTURE)
        cls.raw = loader.load_fixture(cls.FIXTURE)
        cls.uid = cls.entry.get("account_uid") or "316276709526638"

    def test_fixture_integrity(self):
        self.assertEqual(loader.sha256_of(self.raw), self.entry["sha256"])
        self.assertEqual(len(self.raw), self.entry["size"])

    def test_expected_readings_match_manifest(self):
        # 对 parse_conversation_301 的输出做计数
        msgs = cc.parse_conversation_301(self.raw, "some_cid", self.uid)
        got = {
            "total_msgs": len(msgs),
            "with_skey": sum(1 for m in msgs if m.get("skey")),
        }
        for k, want in (self.entry.get("expected") or {}).items():
            with self.subTest(metric=k):
                self.assertEqual(got.get(k), want)

    def test_skey_format(self):
        """skey 必须是 64 hex（AES-256 密钥）。"""
        msgs = cc.parse_conversation_301(self.raw, "some_cid", self.uid)
        for m in msgs:
            skey = m.get("skey")
            if skey:
                self.assertEqual(len(skey), 64)
                int(skey, 16)  # 必须是 hex

    def test_origin_url_unquoted(self):
        """origin_url 中的 \\u0026 已被还原为 &。"""
        msgs = cc.parse_conversation_301(self.raw, "some_cid", self.uid)
        for m in msgs:
            url = m.get("origin_url")
            if url:
                self.assertNotIn("\\u0026", url)
                self.assertIn("&", url)

    def test_pagination_info_present(self):
        """分页信息完整（cursor / total）。"""
        msgs, page = cc._extract_301_page(self.raw)
        self.assertIn("short_id", str(page) or "?")
        # page 应含 '2'(total) 或 '3'(next_cursor)
        self.assertTrue(page.get("2") or page.get("3"))

    def test_sensitivity_truncated_changes_reading(self):
        truncated = self.raw[:256]
        msgs_t = cc.parse_conversation_301(truncated, "cid", self.uid)
        msgs_f = cc.parse_conversation_301(self.raw, "cid", self.uid)
        self.assertNotEqual(len(msgs_t), len(msgs_f))
```

## 4. 样本②：WS 增量帧

### 4.1 来源

WebSocket `wss://frontier-im.douyin.com/ws/v2` 的实时推送帧，经 BCC 的 `CAP_WP_MESSAGE_HOOK_JS` 劫持后推入 `window.__CAP_WP_MESSAGE__.events`（kind=`ws`），由后端 `recv_daemon` 轮询回读、`parse_ws_frame()` 解析。

### 4.2 格式

WS 帧有两种形态：

**A) JSON 帧**（当前唯一可解析）：
```json
{
  "messages": [{
    "conversation_id": "0:1:uid_a:uid_b",
    "conversation_short_id": "...",
    "conversation_type": 1,
    "peer_uid": "对手uid",
    "sender": "发送者uid",
    "text": "消息内容",
    "msg_id": "12345",
    "client_msg_id": "uuid",
    "sender_nickname": "发送者昵称",
    "ts": 1690000000,
    "msg_type": "text"
  }]
}
```
或嵌套 `{"data": {"messages": [...]}}`。

**B) 二进制帧**（B64: 前缀 + base64）：protobuf 格式，当前不解析但保留原始数据。

### 4.3 真机录制步骤

**前置条件**：BCC 在线且目标账号已登录抖音私信页，WP hook 已正确注入。

```
Step 1: 确认 WS 连接已建立（frontier-im.douyin.com ws/v2）
Step 2: 让对端好友发送几条私信（消息类型覆盖 text/image/语音/系统通知）
Step 3: 自己回复几条（触发回声帧，验证方向判定 "me" vs "them"）
Step 4: 从 BCC 日志提取 WS 帧原始 body
        或在 recv_daemon 日志搜索 "kind=ws" 定位帧内容
```

**更准确的录制方式**：从 `__CAP_WP_MESSAGE__.events` 读取。

```python
# 在 BCC 浏览器控制台执行：
const ws_events = window.__CAP_WP_MESSAGE__.events
    .filter(e => e.kind === 'ws')
    .map(e => ({kind: 'ws', body: e.body}));

// 序列化保存（后端 recv_daemon 的 process_events 接受同格式）
console.log(JSON.stringify(ws_events, null, 2));
```

**推荐内容**：至少 10 帧，覆盖以下场景：
- 对方发来 text 消息 → role="them"
- 自己发出消息的回声 → role="me"
- 图片/语音等非纯文本消息（验证 `_norm_msg` 的字段保真）
- 系统通知（验证 `_norm_msg` 的 text 兜底）

```python
class TestWsFramesReplay(unittest.TestCase):
    """WS 增量帧解析 + 方向判定 + 消息落库回归。"""

    FIXTURE = "ws_frames"
    MY_UID = "316276709526638"

    @classmethod
    def setUpClass(cls):
        if cls.FIXTURE not in loader.list_fixtures():
            raise unittest.SkipTest(f"尚未录制 {cls.FIXTURE} 样本")
        cls.entry = loader.describe(cls.FIXTURE)
        cls.raw = loader.load_fixture(cls.FIXTURE)
        cls.events = json.loads(cls.raw)
        cls.msgs = []
        for ev in cls.events:
            cls.msgs.extend(parse_ws_frame(ev["body"], cls.MY_UID))

    def test_total_messages_match_expected(self):
        expected = (self.entry.get("expected") or {}).get("total_messages")
        if expected:
            self.assertEqual(len(self.msgs), expected)

    def test_role_me_exists(self):
        roles = [m["role"] for m in self.msgs]
        self.assertIn("me", roles)

    def test_role_them_exists(self):
        roles = [m["role"] for m in self.msgs]
        self.assertIn("them", roles)

    def test_direction_no_change_on_uid_change(self):
        """方向判定应基于 msg sender vs my_uid，不受 'me'/'them' 字面影响。"""
        # 用不同的 my_uid 重跑，me/them 比例会变
        uid_alt = "999999999999999"
        msgs_alt = []
        for ev in self.events:
            msgs_alt.extend(parse_ws_frame(ev["body"], uid_alt))
        roles_alt = set(m["role"] for m in msgs_alt)
        self.assertTrue(roles_alt, "换 uid 仍应能解析")

    def test_all_frames_parseable(self):
        for i, ev in enumerate(self.events):
            with self.subTest(frame=i):
                # binary B64 帧可能返回空列表，不应抛异常
                try:
                    msgs = parse_ws_frame(ev["body"], self.MY_UID)
                except Exception as e:
                    self.fail(f"帧 {i} 解析异常: {e}")

    def test_conv_id_format(self):
        for m in self.msgs:
            self.assertRegex(str(m.get("conv_id", "")),
                             r"^(\d+:\d+:\d+:\d+|\d+)$",
                             f"conv_id 格式异常: {m.get('conv_id')}")

    # --- 消息落库回归（可选，需要沙箱 DB）---
    def test_insert_into_sandbox_db(self):
        """WS 解析结果可写入临时 DB，不抛异常。"""
        import tempfile, sqlite3
        db = tempfile.mktemp(suffix=".db")
        con = sqlite3.connect(db)
        # 建表（简版）
        con.execute("CREATE TABLE IF NOT EXISTS dm_messages("
                    "account,conv_id,role,text,msg_type,extra,ts,msg_id)")
        for m in self.msgs:
            con.execute(
                "INSERT INTO dm_messages(account,conv_id,role,text,msg_type,extra,ts,msg_id)"
                " VALUES(?,?,?,?,?,?,?,?)",
                ("acct_A", m["conv_id"], m["role"], m["text"], "text", "{}",
                 m["ts"], str(m.get("msg_id") or "")))
        con.commit()
        cnt = con.execute("SELECT COUNT(*) FROM dm_messages").fetchone()[0]
        self.assertEqual(cnt, len(self.msgs))
        os.unlink(db)
```

## 5. 样本③：IndexedDB 昵称表

### 5.1 来源

抖音前端浏览器 IndexedDB 数据库，位于 `<用户自身 uid>_user` 数据库的 `user` 对象存储。

每条记录：`key` = 对端用户的 sec_uid，`value` = `{uid, nickname, sec_uid, avatar_small{uri, url_list}, avatar_thumb, ...}`。

### 5.2 格式

当前已有完整 JS 读取逻辑 `CAP_IDB_USERINFO_JS`，返回 JSON：
```json
{
  "db": "316276709526638_user",
  "total": 44,
  "users": {
    "1234567890123456789": {
      "uid": "1234567890123456789",
      "nickname": "昵称_01",
      "sec_uid": "MS4wLjAB...",
      "avatar_uri": "https://example.invalid/redacted",
      "avatar": "https://example.invalid/redacted",
      "avatar_urls": ["..."],
      "unique_id": "",
      "short_id": "",
      "signature": "",
      "account_cert": {}
    }
    // ... 44 条
  }
}
```

**关键字段**：`users[uid].uid`（数字，与首包 peer_uid 同一体系）+ `nickname`（展示用昵称）。

实测成绩（2026-09-15 实机落地）：`首包 peer_uid ∩ IDB uid = 44/44 = 100%`。

### 5.3 真机录制步骤

**方式 A**（推荐，直接复用现成端点）：

```
Step 1: BCC 在线，目标账号已登录抖音私信页
Step 2: 确保昵称捕获已完成（BCC 预热 / capture_all 跑过一轮）
Step 3: 请求 BCC 的 /userinfo_idb 端点：
  curl -X POST http://127.0.0.1:<bport>/userinfo_idb \
    -H "Content-Type: application/json" \
    -d '{"wait": 5}' \
    -o idb_userinfo.json
Step 4: 确认 users 字段非空且 total≈期望的会话数
```

**方式 B**（纯手动，不依赖 BCC）：

```
Step 1: 打开抖音私信页 douyin.com/chat
Step 2: 打开浏览器 DevTools → Application → IndexedDB
Step 3: 找到 <uid>_user 数据库 → user 对象存储
Step 4: 右键 → Save as JSON（或逐条导出）
```

## 6. 样本④：DOM 会话列表快照

### 6.1 来源

浏览器 DOM 元素 `[class*=conversationConversationItemwrapper]`，由 `CAP_DOM_SWEEP_JS` 逐屏扫描。

### 6.2 格式

每屏抓取为一个 JSON 数组：
```json
[
  {"nickname": "昵称_01", "avatar": "https://...", "desc": "最后一条消息预览"},
  {"nickname": "昵称_02", "avatar": "", "desc": "你好"},
  ...
]
```

现有 JS 代码：
```javascript
const items = document.querySelectorAll('[class*=conversationConversationItemwrapper]');
for (const it of items) {
  const t = it.querySelector('[class*=ConversationItemtitle]');
  const nick = (t.innerText || '').split('\n')[0].trim();
  const img = it.querySelector('img');
  const d = it.querySelector('[class*=ConversationItemDescleft]');
  const desc = d ? d.innerText.replace(/\s+/g, ' ').trim().slice(0, 120) : '';
  out.push({nickname: nick, avatar: img ? img.src : '', desc: desc});
}
```

### 6.3 什么是「虚拟列表滚动累积回归」

BCC 昵称捕获的核心流程（`browser_daemon.py` 的 `capture_userinfo`）：

```
for _round in range(40):                    # 最多 40 屏
    # 1) 点击本屏所有可见会话（触发 im/user/info）
    await _click_all()
    
    # 2) 用 CAP_DOM_SWEEP_JS 抓取当前 DOM
    items = await page.evaluate(CAP_DOM_SWEEP_JS)
    
    # 3) 去重累积：nickname 去重 → 加入 _dom_seen 字典
    for it in items:
        if it.nickname not in _dom_seen:
            _dom_seen[it.nickname] = {nickname, avatar, desc}
    
    # 4) 早退判据：连续 3 轮无新增昵称 → break
    if _dom_total <= _prev:
        _stall += 1
        if _stall >= 3: break
    
    # 5) 平滑滚动到下一屏
    c.scrollTop += c.clientHeight * random(0.65, 0.95)
```

**回归风险**：
- 虚拟列表（Virtual List）在滚动后**复用 DOM 元素**（不是 append 新元素），同一昵称可能在不同屏出现
- `_dom_seen` 以 nickname 去重 —— 若昵称重复（同名不同人）会误认为同一人
- 滚动步长变化可能导致某些会话被跳过（尤其是中间部分）
- 早退判据误判：3 轮无新增就停，但还有更多会话没加载到

**回放方案的本质**：录制多屏快照，输入给累积算法，验证输出集合正确，且不存在「缺了一屏数据导致集合错误」的回归。

### 6.4 真机录制步骤

```

Step 2: 打开 DevTools → Console
Step 3: 开始录制一个数组 `window.__DOM_SNAPSHOTS__ = []`
Step 4: 执行以下循环（模拟 BCC 的滚动逻辑）：
  for (let i = 0; i < 40; i++) {
```

```python
        seen = {}
        for s in snapshots:
            for it in s.get("items", []):
                n = it.get("nickname", "")
                if n and n not in seen:
                    seen[n] = it
        return seen

            nicks = [it.get("nickname", "") for it in s.get("items", [])]
            self.assertEqual(len(nicks), len(set(nicks)),
                             f"第 {i} 屏含重复昵称")

    def test_accumulation_price(self):
        """验证累积结果的 nickname/desc 完整性。"""
        seen = self.accumulate(self.raw)
        for n, v in seen.items():
            with self.subTest(nickname=n):
                self.assertIn("nickname", v)
                self.assertIn("desc", v, "desc 不应为空")
                self.assertIn("avatar", v)

    def test_virtual_list_scroll_coverage(self):
        """模拟 BCC 滚动累积 + 早退判据。"""
        seen = {}
        prev = -1
        stall = 0
        early_exit = None

        for i, s in enumerate(self.raw):
            for it in s.get("items", []):
                n = it.get("nickname", "")
                if n and n not in seen:
                    seen[n] = it

            cur = len(seen)
            if prev >= 0 and cur <= prev:
                stall += 1
                if stall >= 3:
                    early_exit = i
                    break

                stall = 0
            prev = cur

        expected_accum = (self.entry.get("expected") or {}).get("accumulated_unique")
        if expected_accum:

        seen = {}
        prev = -1
        stall = 0
        early_round = None

        for i, s in enumerate(self.raw):
            for it in s.get("items", []):
                n = it.get("nickname", "")
                if n and n not in seen:
                    seen[n] = it
            cur = len(seen)
            if early_round is not None:
                # 早退之后的轮次仍可能出新昵称

            if prev >= 0 and cur <= prev:

                if stall >= 3:
                    early_round = i
            else:
                stall = 0
            prev = cur

        if early_round:
            after = self.accumulate(self.raw[early_round + 1:])
            early_seen = self.accumulate(self.raw[:early_round + 1])
            new_after = {k for k in after if k not in early_seen}
            # 允许少量新增（末屏补漏），但不应有大量遗漏
            self.assertLess(len(new_after), len(after) * 0.1,
                            f"早退后仍有 {len(new_after)} 个新昵称出现")
```
